"""Public-API fault injection, valid against both parent and repair."""
from dataclasses import replace
import json
import pytest
from apex.core import Config, Refused, canonical
from apex.data import session
from apex.fixtures import research_demo_document
from apex.ledger import Ledger, read_verified
from apex.research import run_research

def test_real_research_rejects_retained_prefix(tmp_path, monkeypatch):
    doc, plan = research_demo_document()
    days = sorted({session(r["event_epoch"]) for r in doc["observations"]})[:4]
    doc["observations"] = [r for r in doc["observations"] if session(r["event_epoch"]) in days]
    starts = [min(r["event_epoch"] for r in doc["observations"] if session(r["event_epoch"]) == day) for day in days]
    plan = replace(plan, start=starts[0], development_start=starts[1], holdout_start=starts[2],
                   end=starts[3] + 390 * 60, scan_minutes=60, minimum_training_labels=2,
                   training_window=20, minimum_comparison_sessions=1)
    class LoseClose(Ledger):
        def __init__(self, path):
            super().__init__(path)
            self.injected_path = path
        def append(self, kind, epoch, payload):
            row = super().append(kind, epoch, payload)
            if kind == "RUN_CLOSE":
                p = self.injected_path
                p.write_bytes(p.read_bytes().splitlines(keepends=True)[0])
            return row
    monkeypatch.setattr("apex.research.Ledger", LoseClose)
    out, returned = tmp_path / "research", None
    try:
        with pytest.raises(Refused):
            returned = run_research(canonical(doc).encode(), out, plan=plan,
                                    config=Config(variance="ewma", paths=100))
    finally:
        rows = read_verified(out / "ledger.jsonl")
        complete = out / "COMPLETE"
        print("OBSERVED " + json.dumps({"producer_returned": returned is not None,
              "summary_exists": (out / "summary.json").exists(),
              "complete_exists": complete.exists(), "failed_exists": (out / "FAILED.json").exists(),
              "retained_rows": len(rows), "last_kind": rows[-1]["kind"],
              "complete_names_prefix": complete.exists() and complete.read_text() == rows[-1]["hash"],
              "real_model_path_arrays": len(list(out.glob("*.npy")))}, sort_keys=True))
